import numpy as np

def simulate_phosphene_rendering(current_map, grid_size=8, phosphene_sigma=1.5):
    # Render simulated phosphene visual field from current array
    output_field = np.zeros((64, 64))
    for r in range(grid_size):
        for c in range(grid_size):
            val = current_map[r, c]
            cy, cx = int((r+0.5)*8), int((c+0.5)*8)
            y, x = np.ogrid[:64, :64]
            gaussian = val * np.exp(-((x - cx)**2 + (y - cy)**2) / (2 * phosphene_sigma**2))
            output_field += gaussian
    return output_field
